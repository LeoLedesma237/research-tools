2. Specifying Between-Trial Variability DDMs

2.1 Creating a custom function to generate between-trial variability DDMs

# Create a custom function that will help in creating multiple between-trial variability DDM
make_designDDMGNG <- function(formula, LT = .250) {
  
  design(
    model = DDMGNG, 
    factors = list( subjects = 1, S = c("ng", "go"), E = c("spd", "acc") ), 
    Rlevels = c("TO", "RESP"), 
    LT = LT, 
    formula = formula, 
    functions = list(
      TIMEOUT = function(d) rep(1.5, nrow(d)), 
      Rnogo = function(d) factor(rep("TO", nrow(d)), levels = c("TO", "RESP")), 
      Rgo = function(d) factor(rep("RESP", nrow(d)), levels = c("TO", "RESP")))
  )
}

2.2 Creating the between-trial variability DDM designs

# Create a formula object with the WDM parameter specification
WDM_formula <- list(a ~ E, 
                    v ~ S * E, 
                    t0 ~ E, 
                    Z ~ E)

# Specifying GNG DDM designs for estimating between-trial variability parameters (single)
designGNG_SZ <- make_designDDMGNG(formula = c(WDM_formula, list = (SZ ~ 1)))
designGNG_sv <- make_designDDMGNG(formula = c(WDM_formula, list = (sv ~ 1)))
designGNG_st0 <- make_designDDMGNG(formula = c(WDM_formula, list = (st0 ~ 1)))

# Specifying GNG DDM designs for estimating between-trial variability parameters (multiple)
designGNG_SZ_sv <- make_designDDMGNG(formula = c(WDM_formula, list(SZ ~ 1, sv ~ 1)))
designGNG_SZ_st0 <- make_designDDMGNG(formula = c(WDM_formula, list(SZ ~ 1, st0 ~ 1)))
designGNG_sv_st0 <- make_designDDMGNG(formula = c(WDM_formula, list(sv ~ 1, st0 ~ 1)))

# Specifying the full GNG DDM
designGNG_full <- make_designDDMGNG(formula = c(WDM_formula, list(SZ ~ 1, sv ~ 1, st0 ~ 1)))

2.3 Creating priors distributions

# Prior means on EMC2's transformed scale
pmean <- c(
  a       = log(1.31),
  v       = -1.8,
  v_Sgo   = 2.5,
  t0      = log(.38),
  Z       = qnorm(.57),

  # Between-trial variability parameters
  SZ      = qnorm(.38),
  sv      = log(1.2),
  st0     = log(.03)
)

# Prior SDs on EMC2's transformed scale
psd <- c(
  a       = 0.25,
  v       = 2.5,
  v_Sgo   = 2.5,
  t0      = 0.31,
  Z       = 0.37,

  # Between-trial variability parameters
  SZ      = 0.75,
  sv      = 0.70,
  st0     = 0.90
)

# Creating the specific prior distributions for each between-trial variability DDM
priorGNG_SZ <- prior(designGNG_SZ, pmean = pmean, psd = psd, type = "single")
priorGNG_sv <- prior(designGNG_sv, pmean = pmean, psd = psd, type = "single")
priorGNG_st0 <- prior(designGNG_st0, pmean = pmean, psd = psd, type = "single")

priorGNG_SZ_sv <- prior(designGNG_SZ_sv, pmean = pmean, psd = psd, type = "single")
priorGNG_SZ_st0 <- prior(designGNG_SZ_st0, pmean = pmean, psd = psd, type = "single")
priorGNG_sv_st0 <- prior(designGNG_sv_st0, pmean = pmean, psd = psd, type = "single")

priorGNG_full <- prior(designGNG_full, pmean = pmean, psd = psd, type = "single")
