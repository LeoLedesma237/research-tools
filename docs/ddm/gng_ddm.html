2. Generating a Cumulative Defective Plot

2.1 Creating RT quantiles

# Specify the RT quantiles
rt_probs <- seq(0.1, 0.9, by = 0.2)

# Calculate RT quantiles separately for Go and No-Go responses
rt_quantiles <- gng %>%
  filter(R == "RESP") %>%
  group_by(S) %>%
  reframe(
    rt = quantile(rt, probs = rt_probs, na.rm = TRUE),
    quantile = rt_probs
  )

# Show the reaction time quantiles for go and ng trials

2.2 Baking Accuracy into the RT Quantiles

# Calculate the probability of responding to each stimulus
response_probs <- gng %>%
  group_by(S) %>%
  reframe(
    p_response = mean(R == "RESP")
  )

# Convert the conditional RT quantiles to defective quantiles
defective_quantiles <- rt_quantiles %>%
  left_join(response_probs, by = "S") %>%
  mutate(
    defective_quantile = quantile * p_response
  )

# Plot the defective RT distributions

  ggplot(aes(x = rt, y = defective_quantile, color = S, group = S)) +
  geom_point() +

3. Fitting a Drift Diffusion Model (DDM)

3.1 Preparing the data for model fitting

# Introduce a subjects variable
gng <- mutate(gng, subjects = "001") %>% 
  relocate(subjects, .before = trial_num)

# convert variables to factors
gng <- mutate(gng,
              S = factor(S),
              R = factor(R))

# Convert reaction time into seconds
gng <- mutate(gng,
              rt = rt/1000)

# Specify the reference levels in the dataset
gng <- gng %>%
  mutate(
    R = factor(R, levels = c("TO", "RESP")),
    S = factor(S, levels = c("ng", "go"))
  )

3.2 Specifying a Design and Model

# Specifying the design and model object
dGNG <- design(
  formula = list(v  ~ 0 + S, a  ~ 1, Z  ~ 1, t0 ~ 1, sv ~ 1),
  factors = list(
    subjects = "001",
    S = c("ng", "go")),
  Rlevels = c("TO", "RESP"),
  model = DDMGNG,
  functions = list(
    TIMEOUT = function(d) rep(2, nrow(d)),
    Rnogo  = function(d) factor(rep("TO",   nrow(d)), levels = c("TO", "RESP")),
    Rgo    = function(d) factor(rep("RESP", nrow(d)), levels = c("TO", "RESP"))
  )
)
