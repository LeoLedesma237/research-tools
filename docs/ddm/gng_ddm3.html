Fitting Between-Trial Variability Parameters to GNG data

============================================================

1. GNG DDM + starting-point variability (SZ)

============================================================

designGNG_SZ <- design( model = DDMGNG, factors = list( subjects = 1, S = c(“ng”, “go”), E = c(“spd”, “acc”) ), Rlevels = c(“TO”, “RESP”), LT = LT, formula = list( a ~ E, v ~ S * E, t0 ~ E, Z ~ E, SZ ~ 1 ), functions = list( TIMEOUT = function(d) rep(1.5, nrow(d)), Rnogo = function(d) factor(rep(“TO”, nrow(d)), levels = c(“TO”, “RESP”)), Rgo = function(d) factor(rep(“RESP”, nrow(d)), levels = c(“TO”, “RESP”)) ) )

gng_args <- list( model = DDMGNG, factors = list( subjects = 1, S = c(“ng”, “go”), E = c(“spd”, “acc”) ), Rlevels = c(“TO”, “RESP”), LT = LT, functions = list( TIMEOUT = function(d) rep(1.5, nrow(d)), Rnogo = function(d) factor(rep(“TO”, nrow(d)), levels = c(“TO”, “RESP”)), Rgo = function(d) factor(rep(“RESP”, nrow(d)), levels = c(“TO”, “RESP”)) ) )

designGNG_SZ <- do.call( design, c( gng_args, list(formula = list( a ~ E, v ~ S * E, t0 ~ E, Z ~ E, SZ ~ 1 )) ) )

5. Full GNG DDM: SZ + sv + st0

============================================================

designGNG_full <- design( model = DDMGNG, factors = list( subjects = 1, S = c(“ng”, “go”), E = c(“spd”, “acc”) ), Rlevels = c(“TO”, “RESP”), LT = LT, formula = list( a ~ E, v ~ S * E, t0 ~ E, Z ~ E, SZ ~ 1, sv ~ 1, st0 ~ 1 ), functions = list( TIMEOUT = function(d) rep(1.5, nrow(d)), Rnogo = function(d) factor(rep(“TO”, nrow(d)), levels = c(“TO”, “RESP”)), Rgo = function(d) factor(rep(“RESP”, nrow(d)), levels = c(“TO”, “RESP”)) ) )

Create priors using the default ones for the between-trial variabilities

priorGNG_SZ <- prior( designGNG_SZ, pmean = pmean, psd = psd, type = “single” )

priorGNG_sv <- prior( designGNG_sv, pmean = pmean, psd = psd, type = “single” )

priorGNG_SZ_sv <- prior( designGNG_SZ_sv, pmean = pmean, psd = psd, type = “single” )

priorGNG_full <- prior( designGNG_full, pmean = pmean, psd = psd, type = “single” )

Creating the EMC objects

sDDM_GNG_combined_SZ <- make_emc(data = gng_combined, design = designGNG_SZ, type=“single”, prior=priorGNG_SZ)

sDDM_GNG_combined_sv <- make_emc(data = gng_combined, design = designGNG_sv, type=“single”, prior=priorGNG_sv)

sDDM_GNG_combined_SZ_sv <- make_emc(data = gng_combined, design = designGNG_SZ_sv, type=“single”, prior=priorGNG_SZ_sv)

sDDM_GNG_combined_full <- make_emc(data = gng_combined, design = designGNG_full, type=“single”, prior=priorGNG_full)

Fitting the models

fit_SZ <- fit(sDDM_GNG_combined_SZ) fit_sv <- fit(sDDM_GNG_combined_sv) fit_SZ_sv <- fit(sDDM_GNG_combined_SZ_sv) fit_full <- fit(sDDM_GNG_combined_full)

Checking model convergence

check(fit_SZ) check(fit_sv) check(fit_SZ_sv) check(fit_full)

Creating predicted datasets

ppSZ<- predict(fit_SZ) ppsv<- predict(fit_sv) ppSZ_sv<- predict(fit_SZ_sv) ppfull<- predict(fit_full)

Creating defective distribution plots

plot_cdf(gng_combined, ppSZ, factors = c(“S”)) plot_cdf(gng_combined, ppsv, factors = c(“S”)) plot_cdf(gng_combined, ppSZ_sv, factors = c(“S”))

plot_cdf(gng_combined, ppfull, factors = c(“S”)) plot_cdf(gng_combined, ppfull, factors = c(“S”, “E”))

Print the parameter estimates for the combined GNG data

credint(fit_full, selection=“alpha”, map=TRUE)

make_gng_design <- function(formula) {
